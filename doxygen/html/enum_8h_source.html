54// Returns true if T is an enum annotated with fmt::as_identifiers.
55template <typename T, typename U = remove_cvref_t<T>>
56consteval auto use_identifiers() -> bool {
57 if constexpr (!std::is_enum<U>::value) {
58 return false;
59 } else {
60 return !std::meta::annotations_of_with_type(^^U, ^^as_identifiers_t)
61 .empty();
62 }
63}

65// Returns the underlying value of `value` converted to uint64_t. Negative
66// values wrap around, so the difference of two such values is the distance
67// between them.
68template <typename E> constexpr auto to_uint64(E value) -> uint64_t {
69 return static_cast<uint64_t>(static_cast<underlying_t<E>>(value));
70}

72// Returns the smallest enumerator value of E or 0 if E has no enumerators.
73template <typename E> consteval auto min_enumerator() -> E {
74 auto enumerators = std::meta::enumerators_of(^^E);
75 if (enumerators.empty()) return E();
76 auto min = std::meta::extract<E>(enumerators[0]);
77 for (std::meta::info e : enumerators) {
78 auto value = std::meta::extract<E>(e);
79 if (value < min) min = value;
80 }
81 return min;
82}

84// Returns the size of a table that maps the distance from the smallest
85// enumerator value of E to an identifier, or 0 if the values are too sparse
86// for such a table to be worthwhile.
87template <typename E> consteval auto identifier_table_size() -> size_t {
88 auto enumerators = std::meta::enumerators_of(^^E);
89 if (enumerators.empty()) return 0;
90 auto min = to_uint64(min_enumerator<E>());
91 auto span = uint64_t();
92 for (std::meta::info e : enumerators)
93 span = max_of(span, to_uint64(std::meta::extract<E>(e)) - min);
94 // Require the density of at least 70% to limit the size of the table.
95 // Comparing with the limit instead of adding one to span avoids overflow.
96 auto limit = uint64_t(enumerators.size()) * 10 / 7;
97 return span < limit ? static_cast<size_t>(span) + 1 : 0;
98}

100template <typename E, size_t N = identifier_table_size<E>()>
101consteval auto make_identifier_table() -> std::array<std::string_view, N> {
102 auto ids = std::array<std::string_view, N>();
103 auto min = to_uint64(min_enumerator<E>());
104 for (std::meta::info e : std::meta::enumerators_of(^^E)) {
105 auto i = static_cast<size_t>(to_uint64(std::meta::extract<E>(e)) - min);
106 // identifier_of returns a view of a string with static storage duration so
107 // it doesn't need to be copied. Keep the identifier of the first
108 // enumerator with this value.
109 if (ids[i].size() == 0) ids[i] = std::meta::identifier_of(e);
110 }
111 return ids;
112}

114// Identifiers of enumerators of E indexed by the distance from the smallest
115// enumerator value with empty string views in the holes.
116template <typename E>
117inline constexpr auto identifier_table = make_identifier_table<E>();

119// Returns the size of the hash table that maps enumerator values of E to
120// identifiers. It is the smallest power of two that keeps the load factor at or
121// below 0.5, which guarantees a free slot and therefore terminates the search.
122template <typename E> consteval auto identifier_map_size() -> size_t {
123 auto size = size_t(1);
124 while (size < std::meta::enumerators_of(^^E).size() * 2) size *= 2;
125 return size;
126}
127
128// Returns the index of the first slot to probe for `value`. The bits of the
129// underlying value are mixed because enumerator values are usually small and
130// only the low bits of the hash are used.
131template <typename E, size_t N = identifier_map_size<E>()>
132constexpr auto identifier_slot(E value) -> size_t {
133 auto h = to_uint64(value);
134 h ^= h >> 33;
135 h *= 0xff51afd7ed558ccd;
136 h ^= h >> 33;
137 return static_cast<size_t>(h & (N - 1));
138}

140template <typename E, size_t N = identifier_map_size<E>()>
141consteval auto make_identifier_map()
142 -> std::array<std::pair<E, std::string_view>, N> {
143 auto map = std::array<std::pair<E, std::string_view>, N>();
144 for (std::meta::info e : std::meta::enumerators_of(^^E)) {
145 auto value = std::meta::extract<E>(e);
146 for (auto i = identifier_slot(value);; i = (i + 1) & (N - 1)) {
147 if (map[i].second.size() != 0) {
148 // Keep the identifier of the first enumerator with this value.
149 if (map[i].first == value) break;
150 continue; // The slot is taken by another value; probe the next one.
151 }
152 map[i] = {value, std::meta::identifier_of(e)};
153 break;
154 }
155 }
156 return map;
157}

159// Identifiers of enumerators of E in an open-addressed hash table with linear
160// probing and empty string views in the free slots.
161template <typename E>
162inline constexpr auto identifier_map = make_identifier_map<E>();
163
164// Returns the identifier of the first enumerator of E equal to value or an
165// empty string view if there is no such enumerator.
166template <typename E>
167constexpr auto identifier_of(E value) -> std::string_view {
168 constexpr size_t table_size = identifier_table_size<E>();
169 if constexpr (table_size != 0) {
170 // Values outside of the table wrap around and are rejected by the check.
171 auto i = to_uint64(value) - to_uint64(min_enumerator<E>());
172 return i < table_size ? identifier_table<E>[static_cast<size_t>(i)]
173 : std::string_view();
174 } else {
175 constexpr size_t map_size = identifier_map_size<E>();
176 // A free slot terminates the search and its empty identifier is the result.
177 for (auto i = identifier_slot(value);; i = (i + 1) & (map_size - 1)) {
178 const auto& entry = identifier_map<E>[i];
179 if (entry.second.size() == 0 || entry.first == value) return entry.second;
180 }
181 }
182}

186// A formatter for enums annotated with fmt::as_identifiers.
187template <typename E>
188struct formatter<E, char, enable_if_t<detail::use_identifiers<E>()>> {
189 private:
190 formatter<string_view> impl_;
191
192 public:
193 FMT_CONSTEXPR auto parse(parse_context<char>& ctx) -> const char* {
194 return impl_.parse(ctx);
195 }
196
197 template <typename FormatContext>
198 auto format(E value, FormatContext& ctx) const -> decltype(ctx.out()) {
199 auto id = detail::identifier_of(value);
200 if (id.size() != 0) return impl_.format(id, ctx);
201 // Fall back to the underlying value if there is no matching enumerator.
202 // Unary plus applies integral promotion so that `char` and `bool`
203 // underlying types are written in decimal instead of as a character or
204 // "true"/"false".
205 auto buf = memory_buffer();
206 detail::write<char>(appender(buf), +underlying(value));
207 return impl_.format(string_view(buf.data(), buf.size()), ctx);
208 }
209};
