46inline auto write_loc(basic_appender<wchar_t> out, loc_value value,
47 const format_specs& specs, locale_ref loc) -> bool {
48#if FMT_USE_LOCALE
49 auto locale = loc.get<std::locale>(); // Workaround a gcc 14 bug (#4876).
50 auto& numpunct = std::use_facet<std::numpunct<wchar_t>>(locale);
51 auto separator = std::wstring();
52 auto grouping = numpunct.grouping();
53 if (!grouping.empty()) separator = std::wstring(1, numpunct.thousands_sep());
54 return value.visit(loc_writer<wchar_t>{out, specs, separator, grouping, {}});
55#endif
56 ignore_unused(out, value, specs, loc);
57 return false;
58}

95 template <typename S,
96 FMT_ENABLE_IF(
97 std::is_convertible<const S&, basic_string_view<Char>>::value)>
98 FMT_CONSTEVAL FMT_ALWAYS_INLINE basic_fstring(const S& s) : str_(s) {
99 if (FMT_USE_CONSTEVAL)
100 detail::parse_format_string<Char>(s, checker(s, arg_pack()));
101 }
102 template <typename S,
103 FMT_ENABLE_IF(std::is_base_of<detail::compile_string, S>::value&&
104 std::is_same<typename S::char_type, Char>::value)>
105 FMT_ALWAYS_INLINE basic_fstring(const S&) : str_(S()) {
106 FMT_CONSTEXPR auto sv = basic_string_view<Char>(S());
107 FMT_CONSTEXPR int ignore =
108 (parse_format_string(sv, checker(sv, arg_pack())), 0);
109 detail::ignore_unused(ignore);
110 }
111 basic_fstring(runtime_format_string<Char> fmt) : str_(fmt.str) {}
112
113 FMT_DEPRECATED operator basic_string_view<Char>() const { return str_; }
114 auto get() const -> basic_string_view<Char> { return str_; }
115};
