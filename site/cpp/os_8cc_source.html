75class system_message {
76 system_message(const system_message&) = delete;
77 void operator=(const system_message&) = delete;
78
79 unsigned long result_;
80 wchar_t* message_;
81
82 static bool is_whitespace(wchar_t c) noexcept {
83 return c == L' ' || c == L'\n' || c == L'\r' || c == L'\t' || c == L'\0';
84 }
85
86 public:
87 explicit system_message(unsigned long error_code)
88 : result_(0), message_(nullptr) {
89 result_ = FormatMessageW(
90 FORMAT_MESSAGE_ALLOCATE_BUFFER | FORMAT_MESSAGE_FROM_SYSTEM |
91 FORMAT_MESSAGE_IGNORE_INSERTS,
92 nullptr, error_code, MAKELANGID(LANG_NEUTRAL, SUBLANG_DEFAULT),
93 reinterpret_cast<wchar_t*>(&message_), 0, nullptr);
94 if (result_ != 0) {
95 while (result_ != 0 && is_whitespace(message_[result_ - 1])) {
96 --result_;
97 }
98 }
99 }
100 ~system_message() { LocalFree(message_); }
101 explicit operator bool() const noexcept { return result_ != 0; }
102 operator fmt::basic_string_view<wchar_t>() const noexcept {
103 return fmt::basic_string_view<wchar_t>(message_, result_);
104 }
105};

108 public:

236void file::close() {
237 if (fd_ == -1) return;
238 // Don't retry close in case of EINTR!
239 // See http://linux.derkeiler.com/Mailing-Lists/Kernel/2005-09/3000.html

246auto file::size() const -> long long {
247# ifdef _WIN32
248 // Use GetFileSize instead of GetFileSizeEx for the case when _WIN32_WINNT
249 // is less than 0x0500 as is the case with some default MinGW builds.
250 // Both functions support large file sizes.
251 DWORD size_upper = 0;
252 HANDLE handle = reinterpret_cast<HANDLE>(_get_osfhandle(fd_));
253 DWORD size_lower = FMT_SYSTEM(GetFileSize(handle, &size_upper));
254 if (size_lower == INVALID_FILE_SIZE) {
255 DWORD error = GetLastError();
256 if (error != NO_ERROR)
257 FMT_THROW(windows_error(error, "cannot get file size"));
258 }
259 unsigned long long long_size = size_upper;
260 return (long_size << sizeof(DWORD) * CHAR_BIT) | size_lower;
261# else
262 using Stat = struct stat;
263 Stat file_stat = Stat();
264 if (FMT_POSIX_CALL(fstat(fd_, &file_stat)) == -1)
265 FMT_THROW(system_error(errno, FMT_STRING("cannot get file attributes")));
266 static_assert(sizeof(long long) >= sizeof(file_stat.st_size),
267 "return type of file::size is not large enough");
268 return file_stat.st_size;
269# endif
270}

288auto file::dup(int fd) -> file {
289 // Don't retry as dup doesn't return EINTR.
290 // http://pubs.opengroup.org/onlinepubs/009695399/functions/dup.html
291 int new_fd = FMT_POSIX_CALL(dup(fd));
292 if (new_fd == -1)
293 FMT_THROW(system_error(
294 errno, FMT_STRING("cannot duplicate file descriptor {}"), fd));
295 return file(new_fd);

315// Don't retry as fdopen doesn't return EINTR.
316# if defined(__MINGW32__) && defined(_POSIX_)

342pipe::pipe() {
343 int fds[2] = {};
344# ifdef _WIN32
345 // Make the default pipe capacity same as on Linux 2.6.11+.
346 enum { DEFAULT_CAPACITY = 65536 };
347 int result = FMT_POSIX_CALL(pipe(fds, DEFAULT_CAPACITY, _O_BINARY));
348# else
349 // Don't retry as the pipe function doesn't return EINTR.
350 // http://pubs.opengroup.org/onlinepubs/009696799/functions/pipe.html
351 int result = FMT_POSIX_CALL(pipe(fds));
352# endif
353 if (result != 0)
354 FMT_THROW(system_error(errno, FMT_STRING("cannot create pipe")));
355 // The following assignments don't throw.
356 read_end = file(fds[0]);
357 write_end = file(fds[1]);
358}
