14static int convert_c_format_args(
15 fmt::basic_format_arg<fmt::format_context>* format_args,
16 const fmt_arg* args, size_t num_args) {
17 if (num_args > max_c_format_args) return fmt_error_invalid_arg;
18
19 for (size_t i = 0; i < num_args; ++i) {
20 switch (args[i].type) {
21 case fmt_int: format_args[i] = args[i].value.int_value; break;
22 case fmt_uint: format_args[i] = args[i].value.uint_value; break;
23 case fmt_bool: format_args[i] = args[i].value.bool_value; break;
24 case fmt_char: format_args[i] = args[i].value.char_value; break;
25 case fmt_float: format_args[i] = args[i].value.float_value; break;
26 case fmt_double: format_args[i] = args[i].value.double_value; break;
27 case fmt_long_double:
28 format_args[i] = args[i].value.long_double_value;
29 break;
30 case fmt_cstring: format_args[i] = args[i].value.cstring; break;
31 case fmt_pointer: format_args[i] = args[i].value.pointer; break;
32 default: return fmt_error_invalid_arg;
33 }
34 }
35 return 0;
36}
